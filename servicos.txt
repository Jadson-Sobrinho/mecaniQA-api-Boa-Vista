package br.com.mecaniQA.api.service;

import java.util.List;
import br.com.mecaniQA.api.config.ConnectionFactory;
import br.com.mecaniQA.api.dto.OrdemServicoDTO;
import br.com.mecaniQA.api.model.OrdemServico;
import br.com.mecaniQA.api.mapper.OrdemServicoMapper;
import br.com.mecaniQA.api.repository.OrdemServicoRepository;
import br.com.mecaniQA.api.exception.*;

public class OrdemServicoService {
    private final OrdemServicoRepository repository;
    private final ConnectionFactory db;
                                                                    
    public OrdemServicoService() {
        this.repository = OrdemServicoRepository.getInstance();
        this.db = ConnectionFactory.getInstance();
    }
    public List<OrdemServicoDTO> listar() { return repository.findAll().stream().map(OrdemServicoMapper::toDTO).toList(); }
    public OrdemServicoDTO buscar(Long id) {
        Validacao.id(id);
        return OrdemServicoMapper.toDTO(repository.findById(id).orElseThrow(() -> new RecursoNaoEncontradoException("Ordem nÃ£o encontrada.")));
    }
    public OrdemServicoDTO criar(OrdemServicoDTO dto) {
        validar(dto);
        OrdemServico ordem = OrdemServicoMapper.toEntity(dto); ordem.setCodigo(null);
        return OrdemServicoMapper.toDTO(repository.save(ordem));
    }
    public OrdemServicoDTO atualizar(Long id, OrdemServicoDTO dto) {
        Validacao.id(id); validar(dto);
        return db.transaction(() -> {
            OrdemServico existente = repository.findByIdForUpdate(id).orElseThrow(() -> new RecursoNaoEncontradoException("Ordem nÃ£o encontrada."));
            existente.setDescricao(dto.getDescricao()); existente.setStatus(dto.getStatus());
            return OrdemServicoMapper.toDTO(repository.update(existente));
        });
    }
    public void excluir(Long id) { Validacao.id(id); repository.deleteById(id); }
    private void validar(OrdemServicoDTO dto) {
        Validacao.required(dto,"Dados"); Validacao.text(dto.getDescricao(),"DescriÃ§Ã£o",2000); Validacao.required(dto.getStatus(),"Status");
    }
}

package br.com.mecaniQA.api.service;

import java.util.List;
import br.com.mecaniQA.api.config.ConnectionFactory;
import br.com.mecaniQA.api.dto.*;
import br.com.mecaniQA.api.model.*;
import br.com.mecaniQA.api.mapper.PedidoPecasMapper;
import br.com.mecaniQA.api.repository.*;
import br.com.mecaniQA.api.exception.*;

public class PedidoPecasService {
    private final PedidoPecasRepository repository;
    private final PecaRepository pecas;
    private final ConnectionFactory db;
                                                                    
    public PedidoPecasService() {
        this.repository = PedidoPecasRepository.getInstance();
        this.pecas = PecaRepository.getInstance();
        this.db = ConnectionFactory.getInstance();
    }
    public List<PedidoPecasDTO> listar() { return repository.findAll().stream().map(PedidoPecasMapper::toDTO).toList(); }
    public PedidoPecasDTO buscar(Long id) {
        Validacao.id(id);
        return PedidoPecasMapper.toDTO(repository.findById(id).orElseThrow(() -> new RecursoNaoEncontradoException("Pedido nÃ£o encontrado.")));
    }
                                                                                               
    public PedidoPecasDTO criar(PedidoPecasDTO dto) {
        Validacao.required(dto,"Dados");
        return PedidoPecasMapper.toDTO(repository.save(new PedidoPecas.Builder().build()));
    }
    public PedidoPecasDTO adicionarItem(Long id, ItemPedidoPecaDTO dto) {
        Validacao.id(id); Validacao.required(dto,"Item"); Validacao.id(dto.getPecaCodigo());
        if (dto.getQuantidade() == null || dto.getQuantidade() <= 0) throw new DadosInvalidosException("Quantidade deve ser maior que zero.");
        return db.transaction(() -> {
            PedidoPecas pedido = repository.findByIdForUpdate(id).orElseThrow(() -> new RecursoNaoEncontradoException("Pedido nÃ£o encontrado."));
            if (pedido.getStatus() != StatusPedidoPecas.EM_ORCAMENTO) throw new DadosInvalidosException("SÃ³ Ã© possÃ­vel adicionar peÃ§as a pedidos EM_ORCAMENTO.");
            Peca peca = pecas.findById(dto.getPecaCodigo()).orElseThrow(() -> new RecursoNaoEncontradoException("PeÃ§a nÃ£o encontrada."));
            ItemPedidoPeca item = new ItemPedidoPeca.Builder().pedido(pedido).peca(peca).quantidade(dto.getQuantidade()).build();
            repository.insertItem(id,item);
            return PedidoPecasMapper.toDTO(repository.findById(id).orElseThrow());
        });
    }
    public PedidoPecasDTO alterarStatus(Long id, PedidoPecasDTO dto) {
        Validacao.id(id); Validacao.required(dto,"Dados"); Validacao.required(dto.getStatus(),"Status");
        return db.transaction(() -> {
            repository.findByIdForUpdate(id).orElseThrow(() -> new RecursoNaoEncontradoException("Pedido nÃ£o encontrado."));
            repository.updateStatus(id,dto.getStatus());
            return PedidoPecasMapper.toDTO(repository.findById(id).orElseThrow());
        });
    }
}
